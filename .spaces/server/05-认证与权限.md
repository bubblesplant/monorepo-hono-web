## 5.2 sessionAuth 中间件

新建 `src/modules/auth/session-auth.ts`：

```ts
import { createMiddleware } from 'hono/factory'
import { AppException } from '../../common/errors/app-error'
import { AUTH_ERRORS } from './auth.errors'
import { sessionService, type AuthEnv } from './session.service'
import { userRepository } from './user.repository'

/** 公开路径（精确匹配），登录/注册/健康检查不需要会话。 */
const PUBLIC_PATHS = new Set(['/auth/register', '/auth/login', '/health'])

/**
 * 全局认证中间件：放行公开路径与预检，校验 Bearer 会话及用户状态，
 * 通过后把身份写入 c.set('auth')。
 */
export const sessionAuth = createMiddleware<AuthEnv>(async (c, next) => {
  // CORS 预检不携带业务 token；公开路径直接放行
  if (c.req.method === 'OPTIONS') return next()
  if (PUBLIC_PATHS.has(c.req.path)) return next()
  // OpenAPI 文档与 UI 放行（如需保护可移除）
  if (c.req.path === '/api-docs' || c.req.path.startsWith('/api-docs/')) return next()

  const rawToken = sessionService.extractBearerToken(c.req.header('authorization'))
  if (!rawToken) throw new AppException(AUTH_ERRORS.SESSION_TOKEN_MISSING)

  const auth = await sessionService.validateAndTouch(sessionService.digest(rawToken))
  if (!auth) throw new AppException(AUTH_ERRORS.SESSION_INVALID)

  const user = await userRepository.findPublicById(auth.userId)
  if (!user) throw new AppException(AUTH_ERRORS.SESSION_INVALID)
  // 账号被停用：403 且不带 Bearer 质询——重新登录解决不了
  if (user.status !== 'active') throw new AppException(AUTH_ERRORS.ACCOUNT_DISABLED)

  c.set('auth', auth)
  return next()
})
```

## 5.3 权限判定服务

新建 `src/modules/access/access.service.ts`：

```ts
import { eq } from 'drizzle-orm'
import type { AccessScope } from 'shared/types'
import { db } from '../../db'
import { accessGrants } from '../../db/schema'

export interface AccessQuery {
  readonly actor: { userId: string; requestId?: string }
  readonly scope: AccessScope
  readonly permission?: string | string[]
  readonly adminOnly?: boolean
}

/**
 * 权限判定（简化 RBAC：直接查 access_grants 的用户权限码集合）。
 * 正式版替换为角色→权限的只读事务查询，接口签名不变。
 */
export class AccessService {
  /** 判定 actor 在 scope 下是否拥有全部指定权限；platform.admin 为超级权限。 */
  async check(query: AccessQuery): Promise<boolean> {
    const rows = await db
      .select({ permission: accessGrants.permission })
      .from(accessGrants)
      .where(eq(accessGrants.userId, query.actor.userId))
    const owned = new Set(rows.map((r) => r.permission))

    if (owned.has('platform.admin')) return true
    if (query.adminOnly) return false

    const required = query.permission
      ? Array.isArray(query.permission)
        ? query.permission
        : [query.permission]
      : []
    return required.every((p) => owned.has(p))
  }
}

export const accessService = new AccessService()
```

## 5.4 作用域推导

新建 `src/common/access/scope.ts`：

```ts
import type { Context } from 'hono'
import type { AccessScope } from 'shared/types'
import { z } from 'zod'

const idSchema = z.string().uuid()

/** 按 UUID 校验路由参数；非法参数抛 ZodError，由 onError 转为 400。 */
export function parseIds(c: Context): Record<string, string> {
  const params = c.req.param()
  return Object.fromEntries(
    Object.entries(params).map(([key, value]) => [key, idSchema.parse(value)]),
  )
}

/** 从路由参数推导作用域：有 projectId → project，有 companyId → company，否则 platform。 */
export function scopeFor(c: Context): AccessScope {
  const params = parseIds(c)
  if (params.projectId) {
    return { type: 'project', companyId: params.companyId!, projectId: params.projectId }
  }
  if (params.companyId) {
    return { type: 'company', companyId: params.companyId }
  }
  return { type: 'platform' }
}
```

## 5.5 accessPolicy 中间件工厂

新建 `src/common/access/access-policy.ts`：

```ts
import { createMiddleware } from 'hono/factory'
import type { ScopeType } from 'shared/types'
import { AppException } from '../errors/app-error'
import { ACCESS_ERRORS } from '../../modules/access/access.errors'
import { accessService } from '../../modules/access/access.service'
import { scopeFor } from './scope'
import type { AuthEnv } from '../../modules/auth/session.service'

export interface AccessPolicyDefinition {
  /** authenticated：只需登录；route：按路由参数推导；其余为固定作用域。 */
  readonly scope: ScopeType | 'authenticated' | 'route'
  readonly permission?: string | string[]
  readonly adminOnly?: boolean
}

/** 语法糖：只要求登录身份。 */
export const authenticated = () => accessPolicy({ scope: 'authenticated' })

/**
 * 声明式权限策略中间件，必须在 sessionAuth 之后挂载。
 * permission 中的 {scope} 占位符替换为实际作用域类型，
 * 同一份策略可复用在 platform/company/project 三级路由前缀下。
 */
export function accessPolicy(policy: AccessPolicyDefinition) {
  return createMiddleware<AuthEnv>(async (c, next) => {
    if (c.req.method === 'OPTIONS') return next()

    // 只需登录：sessionAuth 已保证身份
    if (policy.scope === 'authenticated') return next()

    const auth = c.get('auth')
    if (!auth) throw new AppException(ACCESS_ERRORS.FORBIDDEN)

    const scope =
      policy.scope === 'route'
        ? scopeFor(c)
        : policy.scope === 'platform'
          ? ({ type: 'platform' } as const)
          : policy.scope === 'company'
            ? ({ type: 'company', companyId: c.req.param('companyId')! } as const)
            : ({
                type: 'project',
                companyId: c.req.param('companyId')!,
                projectId: c.req.param('projectId')!,
              } as const)

    const permission =
      typeof policy.permission === 'string'
        ? policy.permission.replace('{scope}', scope.type)
        : policy.permission?.map((p) => p.replace('{scope}', scope.type))

    const allowed = await accessService.check({
      actor: { userId: auth.userId, requestId: c.get('requestId') },
      scope,
      ...(permission ? { permission } : {}),
      ...(policy.adminOnly ? { adminOnly: true } : {}),
    })
    if (!allowed) throw new AppException(ACCESS_ERRORS.FORBIDDEN)

    return next()
  })
}
```

## 5.6 auth 路由

新建 `src/modules/auth/auth.routes.ts`：

```ts
import { OpenAPIHono, createRoute, z } from '@hono/zod-openapi'
import argon2 from 'argon2'
import { AppException } from '../../common/errors/app-error'
import { AUTH_ERRORS } from './auth.errors'
import { sessionService, type AuthEnv } from './session.service'
import { userRepository } from './user.repository'

const credentialsSchema = z.object({
  account: z.string().min(3).max(64),
  password: z.string().min(8).max(128),
})

const tokenSchema = z.object({ token: z.string() })
const meSchema = z.object({ user: z.object({ userId: z.string(), sessionId: z.string() }) })

const registerRoute = createRoute({
  method: 'post',
  path: '/register',
  tags: ['auth'],
  request: { body: { content: { 'application/json': { schema: credentialsSchema } } } },
  responses: {
    201: { description: '注册成功', content: { 'application/json': { schema: tokenSchema } } },
  },
})

const loginRoute = createRoute({
  method: 'post',
  path: '/login',
  tags: ['auth'],
  request: { body: { content: { 'application/json': { schema: credentialsSchema } } } },
  responses: {
    200: { description: '登录成功', content: { 'application/json': { schema: tokenSchema } } },
  },
})

const logoutRoute = createRoute({
  method: 'post',
  path: '/logout',
  tags: ['auth'],
  responses: { 204: { description: '已登出' } },
})

const meRoute = createRoute({
  method: 'get',
  path: '/me',
  tags: ['auth'],
  responses: {
    200: { description: '当前会话', content: { 'application/json': { schema: meSchema } } },
  },
})

export const authRoutes = new OpenAPIHono<AuthEnv>()
  .openapi(registerRoute, async (c) => {
    const { account, password } = c.req.valid('json')
    if (await userRepository.findByAccount(account)) {
      throw new AppException(AUTH_ERRORS.ACCOUNT_ALREADY_EXISTS)
    }
    const user = await userRepository.create({ account, passwordHash: await argon2.hash(password) })
    const token = await sessionService.create(user.id)
    return c.json({ token }, 201)
  })
  .openapi(loginRoute, async (c) => {
    const { account, password } = c.req.valid('json')
    const user = await userRepository.findByAccount(account)
    // 用户不存在和密码错误返回同一个错误，防账号枚举
    if (!user || !(await argon2.verify(user.passwordHash, password))) {
      throw new AppException(AUTH_ERRORS.INVALID_CREDENTIALS)
    }
    const token = await sessionService.create(user.id)
    return c.json({ token })
  })
  .openapi(logoutRoute, async (c) => {
    const rawToken = sessionService.extractBearerToken(c.req.header('authorization'))
    if (rawToken) await sessionService.revoke(sessionService.digest(rawToken))
    return c.body(null, 204)
  })
  .openapi(meRoute, (c) => {
    return c.json({ user: c.get('auth') }, 200)
  })
```

> zod-openapi 校验失败默认抛 `ZodError` 风格异常，会被 `onError` 归一化为 `COMMON.VALIDATION_FAILED`，错误出口保持唯一，无需额外 hook。
