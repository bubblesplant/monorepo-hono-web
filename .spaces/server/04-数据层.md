# 04 - 数据层

两层存储，职责分明：

| 存储 | 放什么 | 为什么 |
|---|---|---|
| PostgreSQL（Drizzle） | users、access_grants | 持久、强一致、可关联查询 |
| Redis（ioredis） | 会话 token | 高频读写、TTL 天然过期、可服务端吊销 |

## 4.2 Schema（从简起步）

新建 `src/db/schema.ts`：

```ts
import { pgEnum, pgTable, primaryKey, text, timestamp, uuid } from 'drizzle-orm/pg-core'

export const userStatus = pgEnum('user_status', ['active', 'disabled'])

/** 用户表。 */
export const users = pgTable('users', {
  id: uuid('id').primaryKey().defaultRandom(),
  account: text('account').notNull().unique(),
  passwordHash: text('password_hash').notNull(),
  status: userStatus('status').notNull().default('active'),
  createdAt: timestamp('created_at', { withTimezone: true }).notNull().defaultNow(),
})

/**
 * 权限授予表（教学用简化 RBAC：直接把权限码授给用户）。
 * 正式版替换为 Nest 版 access.schema.ts 的角色→权限模型。
 */
export const accessGrants = pgTable(
  'access_grants',
  {
    userId: uuid('user_id')
      .notNull()
      .references(() => users.id, { onDelete: 'cascade' }),
    permission: text('permission').notNull(),
  },
  (table) => [primaryKey({ columns: [table.userId, table.permission] })],
)
```

建库并推送 schema：

```bash
docker exec -i <pg容器名> createdb -U postgres server_hono   # 或任何方式建库
cd apps/server-hono
pnpm db:push
```

## 4.3 会话服务（Redis）

新建 `src/modules/auth/session.service.ts`：

```ts
import { createHash, randomBytes } from 'node:crypto'
import { Redis } from 'ioredis'
import { env } from '../../config/env'

/** 写入请求上下文的已认证身份。 */
export interface AuthContext {
  readonly userId: string
  readonly sessionId: string
}

/** Hono 环境变量类型，供 c.get('auth') / c.get('requestId') 获得类型提示。 */
export type AuthEnv = {
  Variables: {
    auth: AuthContext
    requestId: string
  }
}

const SESSION_KEY_PREFIX = 'session:'

export class SessionService {
  private readonly redis = new Redis(env.redisUrl, {
    enableOfflineQueue: false,
    connectTimeout: 3000,
    commandTimeout: 2000,
  })

  /** 生成明文 token（仅返回给客户端一次）。 */
  private generateToken(): string {
    return randomBytes(32).toString('base64url')
  }

  /** 对 token 做 SHA-256 摘要；服务端只存摘要，Redis 泄漏也伪造不了会话。 */
  digest(rawToken: string): string {
    return createHash('sha256').update(rawToken).digest('hex')
  }

  /** 从 Authorization 头提取 Bearer token，格式非法返回 null。 */
  extractBearerToken(authorization: string | undefined): string | null {
    if (!authorization) return null
    const match = /^Bearer\s+([A-Za-z0-9._~+/=-]+)$/.exec(authorization.trim())
    return match?.[1] ?? null
  }

  /** 创建会话并返回明文 token。 */
  async create(userId: string): Promise<string> {
    const token = this.generateToken()
    const sessionId = randomBytes(8).toString('hex')
    await this.redis.set(
      SESSION_KEY_PREFIX + this.digest(token),
      JSON.stringify({ userId, sessionId } satisfies AuthContext),
      'EX',
      env.sessionTtlSeconds,
    )
    return token
  }

  /** 校验会话并滑动续期（活跃用户不过期）；不存在返回 null。 */
  async validateAndTouch(digest: string): Promise<AuthContext | null> {
    const key = SESSION_KEY_PREFIX + digest
    const raw = await this.redis.get(key)
    if (!raw) return null
    await this.redis.expire(key, env.sessionTtlSeconds)
    return JSON.parse(raw) as AuthContext
  }

  /** 吊销会话（登出）。 */
  async revoke(digest: string): Promise<void> {
    await this.redis.del(SESSION_KEY_PREFIX + digest)
  }
}

export const sessionService = new SessionService()
```

## 4.4 用户仓储

新建 `src/modules/auth/user.repository.ts`：

```ts
import { eq } from 'drizzle-orm'
import { db } from '../../db'
import { users } from '../../db/schema'

/** 用户表读写。 */
export const userRepository = {
  async findPublicById(id: string) {
    const rows = await db
      .select({ id: users.id, account: users.account, status: users.status })
      .from(users)
      .where(eq(users.id, id))
      .limit(1)
    return rows[0] ?? null
  },

  async findByAccount(account: string) {
    const rows = await db.select().from(users).where(eq(users.account, account)).limit(1)
    return rows[0] ?? null
  },

  async create(input: { account: string; passwordHash: string }) {
    const rows = await db.insert(users).values(input).returning({ id: users.id })
    return rows[0]!
  },
}
```
