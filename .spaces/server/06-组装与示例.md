# 06 - 组装与示例

## 6.1 示例业务模块

新建 `src/modules/companies/companies.routes.ts`——演示三级作用域下同一策略复用：

```ts
import { Hono } from 'hono'
import { zValidator } from '@hono/zod-validator'
import { z } from 'zod'
import { accessPolicy, authenticated } from '../../common/access/access-policy'
import type { AuthEnv } from '../auth/session.service'

const createCompanySchema = z.object({
  name: z.string().min(1).max(100),
})

export const companiesRoutes = new Hono<AuthEnv>()
  // 只需登录
  .get('/mine', authenticated(), (c) => {
    return c.json({ companies: [], userId: c.get('auth').userId })
  })
  // 平台级：要求 platform.company.create 权限
  .post(
    '/',
    accessPolicy({ scope: 'platform', permission: 'platform.company.create' }),
    zValidator('json', createCompanySchema, (result) => {
      if (!result.success) throw result.error
    }),
    (c) => {
      const body = c.req.valid('json')
      return c.json({ id: crypto.randomUUID(), name: body.name }, 201)
    },
  )
  // scope 占位符：同一策略挂在公司级和项目级两个前缀下，
  // {scope} 分别解析为 company.member.read 和 project.member.read
  .get(
    '/:companyId/members',
    accessPolicy({ scope: 'route', permission: '{scope}.member.read' }),
    (c) => {
      return c.json({ companyId: c.req.param('companyId'), members: [] })
    },
  )

export const projectMemberRoutes = new Hono<AuthEnv>().get(
  '/:companyId/projects/:projectId/members',
  accessPolicy({ scope: 'route', permission: '{scope}.member.read' }),
  (c) => {
    return c.json({
      companyId: c.req.param('companyId'),
      projectId: c.req.param('projectId'),
      members: [],
    })
  },
)
```

## 6.2 应用组装

新建 `src/app.ts`：

```ts
import { Hono } from 'hono'
import { requestId } from 'hono/request-id'
import { cors } from 'hono/cors'
import { logger } from 'hono/logger'
import { notFound, onError } from './common/http/on-error'
import { sessionAuth } from './modules/auth/session-auth'
import { authRoutes } from './modules/auth/auth.routes'
import { companiesRoutes, projectMemberRoutes } from './modules/companies/companies.routes'
import type { AuthEnv } from './modules/auth/session.service'

/**
 * 组装应用：中间件顺序即请求处理顺序。
 * requestId → cors → logger → sessionAuth → 业务路由（各自挂 accessPolicy）。
 */
export function createApp() {
  const app = new Hono<AuthEnv>()

  app.use('*', requestId())
  app.use(
    '*',
    cors({
      origin: '*',
      allowHeaders: ['Content-Type', 'Authorization', 'X-Company-Id'],
      exposeHeaders: ['X-Request-Id', 'WWW-Authenticate', 'Retry-After'],
    }),
  )
  app.use('*', logger())

  // 全局认证：公开路径在 session-auth.ts 的 PUBLIC_PATHS 中维护
  app.use('*', sessionAuth)

  app.get('/health', (c) => c.json({ ok: true }))

  app.route('/auth', authRoutes)
  app.route('/companies', companiesRoutes)
  app.route('/companies', projectMemberRoutes)

  app.onError(onError)
  app.notFound(notFound)

  return app
}
```

> 注意 `x-request-id`：Hono 的 `requestId()` 中间件会自动把 ID 写入响应头，onError 里通过 `c.get('requestId')` 读同一个值记日志。

## 6.3 启动入口

新建 `src/main.ts`：

```ts
import { serve } from '@hono/node-server'
import { createApp } from './app'

const port = Number(process.env.PORT ?? 3000)
const hostname = process.env.HOST ?? '0.0.0.0'

const app = createApp()

serve({ fetch: app.fetch, port, hostname }, (info) => {
  console.log(`server listening on http://localhost:${info.port}`)
})

// 优雅退出
for (const signal of ['SIGINT', 'SIGTERM'] as const) {
  process.on(signal, () => {
    console.log(`received ${signal}, shutting down`)
    process.exit(0)
  })
}
```

## 6.4 加载 .env（可选但推荐）

`tsx` 不会自动读 `.env`，加一行即可——把 `src/main.ts` 第一行改为：

```ts
import 'node:process'
```

并在启动命令用 Node 20+ 的 `--env-file`。把 package.json 的 scripts 改为：

```json
"scripts": {
  "dev": "tsx watch --env-file=.env src/main.ts",
  "start": "tsx --env-file=.env src/main.ts",
  "typecheck": "tsc --noEmit"
}
```
