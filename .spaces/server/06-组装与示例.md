# 06 - 组装与 OpenAPI

## 6.1 示例业务模块

新建 `src/modules/companies/companies.routes.ts`——演示 `{scope}` 占位符在三级前缀下复用同一份策略：

```ts
import { OpenAPIHono, createRoute, z } from '@hono/zod-openapi'
import { accessPolicy, authenticated } from '../../common/access/access-policy'
import type { AuthEnv } from '../auth/session.service'

const mineRoute = createRoute({
  method: 'get',
  path: '/mine',
  tags: ['companies'],
  responses: { 200: { description: '我加入的公司' } },
})

const membersRoute = createRoute({
  method: 'get',
  path: '/{companyId}/members',
  tags: ['companies'],
  request: { params: z.object({ companyId: z.string().uuid() }) },
  responses: { 200: { description: '公司成员列表' } },
})

const projectMembersRoute = createRoute({
  method: 'get',
  path: '/{companyId}/projects/{projectId}/members',
  tags: ['companies'],
  request: {
    params: z.object({ companyId: z.string().uuid(), projectId: z.string().uuid() }),
  },
  responses: { 200: { description: '项目成员列表' } },
})

export const companiesRoutes = new OpenAPIHono<AuthEnv>()
  // 只需登录
  .openapi(mineRoute, authenticated(), (c) => {
    return c.json({ companies: [], userId: c.get('auth').userId })
  })
  // {scope} → company.member.read
  .openapi(membersRoute, accessPolicy({ scope: 'route', permission: '{scope}.member.read' }), (c) => {
    return c.json({ companyId: c.req.param('companyId'), members: [] })
  })
  // {scope} → project.member.read
  .openapi(
    projectMembersRoute,
    accessPolicy({ scope: 'route', permission: '{scope}.member.read' }),
    (c) => {
      return c.json({
        companyId: c.req.param('companyId'),
        projectId: c.req.param('projectId'),
        members: [],
      })
    },
  )
```

> 「默认拒绝」约定：业务路由永远挂在 `sessionAuth` 之后，且每个受保护路由必须挂 `accessPolicy`——漏挂在 code review 中视为缺陷（Nest 版靠全局守卫强制，Hono 靠此约定 + review）。

## 6.2 应用组装

新建 `src/app.ts`：

```ts
import { OpenAPIHono } from '@hono/zod-openapi'
import { swaggerUI } from '@hono/swagger-ui'
import { requestId } from 'hono/request-id'
import { cors } from 'hono/cors'
import { logger } from 'hono/logger'
import { notFound, onError } from './common/http/on-error'
import { sessionAuth } from './modules/auth/session-auth'
import { authRoutes } from './modules/auth/auth.routes'
import { companiesRoutes } from './modules/companies/companies.routes'
import type { AuthEnv } from './modules/auth/session.service'

/**
 * 组装应用：中间件顺序即处理顺序。
 * requestId → cors → logger → sessionAuth → 业务路由（各自挂 accessPolicy）。
 */
export function createApp() {
  const app = new OpenAPIHono<AuthEnv>()

  app.use('*', requestId())
  app.use(
    '*',
    cors({
      origin: '*',
      allowHeaders: ['Content-Type', 'Authorization', 'X-Company-Id'],
      exposeHeaders: ['X-Request-Id', 'WWW-Authenticate', 'Retry-After'],
    }),
  )
  app.use('*', logger())
  app.use('*', sessionAuth)

  app.get('/health', (c) => c.json({ ok: true }))

  app.route('/auth', authRoutes)
  app.route('/companies', companiesRoutes)

  // OpenAPI 文档 + Swagger UI
  app.doc('/api-docs/openapi.json', {
    openapi: '3.1.0',
    info: { title: 'server-hono API', version: '0.1.0' },
  })
  app.openAPIRegistry.registerComponent('securitySchemes', 'session', {
    type: 'http',
    scheme: 'bearer',
    description: 'Opaque Session Token',
  })
  app.get('/api-docs', swaggerUI({ url: '/api-docs/openapi.json' }))

  app.onError(onError)
  app.notFound(notFound)

  return app
}

export type AppType = ReturnType<typeof createApp>
```

## 6.3 双运行时入口

新建 `src/main.node.ts`：

```ts
import { serve } from '@hono/node-server'
import { createApp } from './app'
import { env } from './config/env'

const app = createApp()

serve({ fetch: app.fetch, port: env.port, hostname: env.host }, (info) => {
  console.log(`[node] server listening on http://localhost:${info.port}`)
  console.log(`[node] swagger ui: http://localhost:${info.port}/api-docs`)
})

for (const signal of ['SIGINT', 'SIGTERM'] as const) {
  process.on(signal, () => {
    console.log(`received ${signal}, shutting down`)
    process.exit(0)
  })
}
```

新建 `src/main.bun.ts`（可选，验证双平台用）：

```ts
import { createApp } from './app'
import { env } from './config/env'

const app = createApp()

export default {
  port: env.port,
  hostname: env.host,
  fetch: app.fetch,
}

console.log(`[bun] server listening on http://localhost:${env.port}`)
```

> Bun 入口首次使用时先验证 `/auth/register` + `/auth/login`（argon2 是 napi 原生模块，是双平台唯一需要实测的点）。
