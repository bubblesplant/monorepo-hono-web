# 04 - 认证

对应 Nest 版 `SessionAuthGuard`。设计：**Bearer 不透明 token（不是 JWT）+ Redis 滑动续期**。

- token 明文只出现在客户端一次（登录响应），服务端只存 SHA-256 digest → Redis 泄漏也伪造不了会话
- 每次请求 `validateAndTouch` 顺手刷新 TTL → 活跃用户不过期，闲置 7 天自动失效
- 服务端可吊销（删 key 即登出）——这是不透明 token 相对 JWT 的核心优势

## 4.1 认证错误注册表

新建 `src/modules/auth/auth.errors.ts`：

```ts
import type { AppErrorDefinition } from '../../common/errors/app-error'

export const AUTH_ERRORS = {
  SESSION_TOKEN_MISSING: {
    code: 'AUTH.SESSION_TOKEN_MISSING',
    publicMessage: '缺少有效的登录 Token',
    status: 401,
    bearerChallenge: true,
  },
  SESSION_INVALID: {
    code: 'AUTH.SESSION_INVALID',
    publicMessage: '登录已失效，请重新登录',
    status: 401,
    bearerChallenge: true,
  },
  INVALID_CREDENTIALS: {
    code: 'AUTH.INVALID_CREDENTIALS',
    publicMessage: '账号或密码错误',
    status: 422,
  },
  ACCOUNT_ALREADY_EXISTS: {
    code: 'AUTH.ACCOUNT_ALREADY_EXISTS',
    publicMessage: '账号已存在',
    status: 409,
  },
} as const satisfies Record<string, AppErrorDefinition>
```

> 注意 `SESSION_INVALID` 也带 `bearerChallenge`——token 过期/被吊销正是需要客户端重新走登录流程的场景（对照：账号被禁用不在这里声明，见 4.3 的停用分支用 `FORBIDDEN`）。

## 4.2 会话服务

新建 `src/modules/auth/session.service.ts`：

```ts
import { createHash, randomBytes } from 'node:crypto'
import { Redis } from 'ioredis'

/**
 * 写入请求上下文的已认证身份。
 */
export interface AuthContext {
  readonly userId: string
  readonly sessionId: string
}

const SESSION_KEY_PREFIX = 'session:'
const TTL_SECONDS = Number(process.env.SESSION_TTL_SECONDS ?? 60 * 60 * 24 * 7)

/** 声明 Hono 环境变量类型，供 c.get('auth') 获得类型提示。 */
export type AuthEnv = {
  Variables: {
    auth: AuthContext
    requestId: string
  }
}

export class SessionService {
  private readonly redis: Redis

  constructor() {
    this.redis = new Redis(process.env.REDIS_URL ?? 'redis://localhost:6379', {
      lazyConnect: false,
      enableOfflineQueue: false,
      connectTimeout: 3000,
      commandTimeout: 2000,
    })
  }

  /** 生成明文 token（仅返回给客户端一次）。 */
  generateToken(): string {
    return randomBytes(32).toString('base64url')
  }

  /** 对 token 做 SHA-256 摘要；服务端只存摘要。 */
  digest(rawToken: string): string {
    return createHash('sha256').update(rawToken).digest('hex')
  }

  /** 从 Authorization 头提取 Bearer token，格式非法返回 null。 */
  extractBearerToken(authorization: string | undefined): string | null {
    if (!authorization) return null
    const match = /^Bearer\s+([A-Za-z0-9._~+/=-]+)$/.exec(authorization.trim())
    return match?.[1] ?? null
  }

  /** 创建会话并返回明文 token。 */
  async create(userId: string): Promise<string> {
    const token = this.generateToken()
    const digest = this.digest(token)
    const sessionId = randomBytes(8).toString('hex')
    await this.redis.set(
      SESSION_KEY_PREFIX + digest,
      JSON.stringify({ userId, sessionId } satisfies AuthContext),
      'EX',
      TTL_SECONDS,
    )
    return token
  }

  /** 校验会话并滑动续期；不存在返回 null。 */
  async validateAndTouch(digest: string): Promise<AuthContext | null> {
    const key = SESSION_KEY_PREFIX + digest
    const raw = await this.redis.get(key)
    if (!raw) return null
    // 滑动续期：活跃用户的 TTL 重置
    await this.redis.expire(key, TTL_SECONDS)
    return JSON.parse(raw) as AuthContext
  }

  /** 吊销会话（登出）。 */
  async revoke(digest: string): Promise<void> {
    await this.redis.del(SESSION_KEY_PREFIX + digest)
  }
}

export const sessionService = new SessionService()
```

## 4.3 sessionAuth 中间件

新建 `src/modules/auth/session-auth.ts`：

```ts
import { createMiddleware } from 'hono/factory'
import { AppException } from '../../common/errors/app-error'
import { AUTH_ERRORS } from './auth.errors'
import { sessionService, type AuthEnv } from './session.service'

/**
 * 极简用户仓储。真实项目换成 Drizzle 查询，接口保持不变。
 */
const DEMO_USERS = new Map<
  string,
  { id: string; account: string; passwordHash: string; status: 'active' | 'disabled' }
>()

export const userRepository = {
  async findPublicById(id: string) {
    const user = DEMO_USERS.get(id)
    return user ? { id: user.id, account: user.account, status: user.status } : null
  },
  async findByAccount(account: string) {
    return [...DEMO_USERS.values()].find((u) => u.account === account) ?? null
  },
  async create(user: {
    id: string
    account: string
    passwordHash: string
    status: 'active'
  }) {
    DEMO_USERS.set(user.id, user)
    return user
  },
}

/** 公开路径（精确匹配），登录/注册不需要会话。 */
const PUBLIC_PATHS = new Set(['/auth/register', '/auth/login', '/health'])

/**
 * 全局认证中间件：放行公开路径与预检，校验 Bearer 会话及用户状态，
 * 通过后把身份写入 c.set('auth')。
 */
export const sessionAuth = createMiddleware<AuthEnv>(async (c, next) => {
  // CORS 预检不携带业务 token
  if (c.req.method === 'OPTIONS') return next()
  if (PUBLIC_PATHS.has(c.req.path)) return next()

  const rawToken = sessionService.extractBearerToken(c.req.header('authorization'))
  if (!rawToken) throw new AppException(AUTH_ERRORS.SESSION_TOKEN_MISSING)

  const auth = await sessionService.validateAndTouch(sessionService.digest(rawToken))
  if (!auth) throw new AppException(AUTH_ERRORS.SESSION_INVALID)

  const user = await userRepository.findPublicById(auth.userId)
  if (!user) throw new AppException(AUTH_ERRORS.SESSION_INVALID)
  // 账号被停用：403 且不带 Bearer 质询——重新登录也解决不了
  if (user.status !== 'active') {
    throw new AppException({
      code: 'AUTH.ACCOUNT_DISABLED',
      publicMessage: '账号已被停用',
      status: 403,
    })
  }

  c.set('auth', auth)
  return next()
})
```

## 4.4 登录/注册/登出路由

新建 `src/modules/auth/auth.routes.ts`：

```ts
import { Hono } from 'hono'
import { zValidator } from '@hono/zod-validator'
import { randomUUID } from 'node:crypto'
import argon2 from 'argon2'
import { z } from 'zod'
import { AppException } from '../../common/errors/app-error'
import { AUTH_ERRORS } from './auth.errors'
import { sessionService, type AuthEnv } from './session.service'
import { userRepository } from './session-auth'

const credentialsSchema = z.object({
  account: z.string().min(3).max(64),
  password: z.string().min(8).max(128),
})

/** 校验失败统一抛给 onError 处理，保持错误出口唯一。 */
const json = (schema: typeof credentialsSchema) =>
  zValidator('json', schema, (result) => {
    if (!result.success) throw result.error
  })

export const authRoutes = new Hono<AuthEnv>()
  .post('/register', json(credentialsSchema), async (c) => {
    const { account, password } = c.req.valid('json')
    if (await userRepository.findByAccount(account)) {
      throw new AppException(AUTH_ERRORS.ACCOUNT_ALREADY_EXISTS)
    }
    const user = await userRepository.create({
      id: randomUUID(),
      account,
      passwordHash: await argon2.hash(password),
      status: 'active',
    })
    const token = await sessionService.create(user.id)
    return c.json({ token }, 201)
  })
  .post('/login', json(credentialsSchema), async (c) => {
    const { account, password } = c.req.valid('json')
    const user = await userRepository.findByAccount(account)
    // 用户不存在和密码错误返回同一个错误，防账号枚举
    if (!user || !(await argon2.verify(user.passwordHash, password))) {
      throw new AppException(AUTH_ERRORS.INVALID_CREDENTIALS)
    }
    const token = await sessionService.create(user.id)
    return c.json({ token })
  })
  .post('/logout', async (c) => {
    const rawToken = sessionService.extractBearerToken(c.req.header('authorization'))
    if (rawToken) await sessionService.revoke(sessionService.digest(rawToken))
    return c.body(null, 204)
  })
  .get('/me', (c) => {
    return c.json({ user: c.get('auth') })
  })
```

> `POST /register`、`POST /login` 在 `PUBLIC_PATHS` 里，中间件放行；`/me`、`/logout` 需要有效会话。
